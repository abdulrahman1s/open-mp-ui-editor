cmake_minimum_required(VERSION 3.24)
project(open-mp-ui-editor VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
FetchContent_Declare(omp_deps
  GIT_REPOSITORY https://github.com/openmultiplayer/pawn-template.git
  GIT_TAG 28ca281e00a5650d116b8e7a903eba0868586d3a
  GIT_SUBMODULES sdk pawn
  GIT_SUBMODULES_RECURSE TRUE
  SOURCE_SUBDIR unused-template-project)
FetchContent_MakeAvailable(omp_deps)
add_subdirectory(${omp_deps_SOURCE_DIR}/sdk ${omp_deps_BINARY_DIR}/sdk EXCLUDE_FROM_ALL)
set(UI_RUST_LIBRARY "" CACHE FILEPATH "Path to target/<triple>/release/libopen_mp_ui_runtime.a (or .lib)")
if(NOT EXISTS "${UI_RUST_LIBRARY}")
  message(FATAL_ERROR "Build the Rust static library and set UI_RUST_LIBRARY to its absolute path")
endif()
add_library(open-mp-ui-editor SHARED component.cpp)
set_target_properties(open-mp-ui-editor PROPERTIES PREFIX "")
option(UI_TEST_DRIVER "Compile private SDK event injection natives for isolated integration tests" OFF)
if(UI_TEST_DRIVER)
  target_compile_definitions(open-mp-ui-editor PRIVATE UI_TEST_DRIVER=1)
endif()
target_include_directories(open-mp-ui-editor PRIVATE ${omp_deps_SOURCE_DIR}/pawn/source ${omp_deps_SOURCE_DIR}/pawn/source/linux)
target_compile_definitions(open-mp-ui-editor PRIVATE HAVE_STDINT_H=1 PAWN_CELL_SIZE=32)
target_link_libraries(open-mp-ui-editor PRIVATE OMP-SDK ${UI_RUST_LIBRARY})
if(WIN32)
  target_link_libraries(open-mp-ui-editor PRIVATE ws2_32 userenv bcrypt ntdll advapi32)
else()
  target_link_libraries(open-mp-ui-editor PRIVATE dl pthread m)
  target_link_options(open-mp-ui-editor PRIVATE -Wl,--no-undefined)
endif()
install(TARGETS open-mp-ui-editor LIBRARY DESTINATION components RUNTIME DESTINATION components)
